 public interface CarpetaRepository extends JpaRepository<Carpeta, Integer> {
 
     @Query("""
             SELECT c

             AND c.juzgado.id = :juzgadoId
             """)
     Optional<Carpeta> findByExpedienteAndJuzgadoId(String expediente, Integer juzgadoId);
 
 }
